1: # -*- coding: utf-8 -*-
2: """HR 四级分级分类器 —— 用于分辨率消融实验（128 / 256 / 512 哪个够用）。
3: 
4: 设计取舍（为了让"分辨率"成为唯一变量）：
5:   · 训练时**不做数据增强**，只 resize + 归一化
6:   · 固定用数据集自带的 train/val/test 划分（不重新切分，保持与老师协议一致）
7:   · 多个随机种子重复训练，报告 均值 ± 标准差
8: 
9: 用法：
10:     python classify_project/scripts/train_classifier.py --data fundus/HRSG_4_758_square --size 256
11:     python classify_project/scripts/train_classifier.py --data ... --size 256 --seeds 42 52 62
12: """
13: from __future__ import annotations
14: 
15: import argparse
16: import json
17: import os
18: import random
19: import sys
20: import time
21: from pathlib import Path
22: 
23: import numpy as np
24: import torch
25: import torch.nn as nn
26: from PIL import Image
27: from sklearn.metrics import (balanced_accuracy_score, cohen_kappa_score,
28:                              confusion_matrix, f1_score)
29: from torch.utils.data import DataLoader, Dataset, WeightedRandomSampler
30: from torchvision import models, transforms
31: 
32: Image.MAX_IMAGE_PIXELS = None
33: CLASSES = ["normal", "mild", "moderate", "severe"]   # 有序！0<1<2<3
34: NUM_CLASSES = len(CLASSES)
35: 
36: 
37: class FundusFolder(Dataset):
38:     """按 {root}/{split}/{class}/*.jpg 读取。"""
39: 
40:     def __init__(self, root: Path, split: str, size: int, train: bool):
41:         self.items: list[tuple[Path, int]] = []
42:         for ci, cname in enumerate(CLASSES):
43:             d = root / split / cname
44:             if not d.is_dir():
45:                 continue
46:             for f in sorted(d.iterdir()):
47:                 if f.suffix.lower() in (".jpg", ".jpeg", ".png", ".bmp"):
48:                     self.items.append((f, ci))
49:         self.size = size
50:         # 训练/评估用同一套预处理（只是缩放+归一化），保证分辨率是唯一变量
51:         self.tf = transforms.Compose([
52:             transforms.Resize((size, size), antialias=True),
53:             transforms.ToTensor(),
54:             transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225]),
55:         ])
56: 
57:     def __len__(self) -> int:
58:         return len(self.items)
59: 
60:     def __getitem__(self, i: int):
61:         path, label = self.items[i]
62:         with Image.open(path) as im:
63:             im = im.convert("RGB")
64:         return self.tf(im), label
65: 
66: 
67: def build_model(size: int, pretrained: bool = True) -> nn.Module:
68:     weights = models.ResNet18_Weights.IMAGENET1K_V1 if pretrained else None
69:     m = models.resnet18(weights=weights)
70:     m.fc = nn.Linear(m.fc.in_features, NUM_CLASSES)
71:     return m
72: 
73: 
74: def metrics_from_preds(y_true: np.ndarray, y_pred: np.ndarray) -> dict:
75:     severe = CLASSES.index("severe")
76:     rec = {}
77:     for i, c in enumerate(CLASSES):
78:         m = y_true == i
79:         rec[f"recall_{c}"] = float((y_pred[m] == i).mean()) if m.any() else float("nan")
80:     return {
81:         "acc": float((y_true == y_pred).mean()),
82:         "balanced_acc": float(balanced_accuracy_score(y_true, y_pred)),
83:         "macro_f1": float(f1_score(y_true, y_pred, average="macro", zero_division=0)),
84:         "qwk": float(cohen_kappa_score(y_true, y_pred, weights="quadratic")),
85:         "severe_recall": rec["recall_severe"],
86:         "n": int(len(y_true)),
87:         "confusion": confusion_matrix(y_true, y_pred, labels=range(NUM_CLASSES)).tolist(),
88:     }
89: 
90: 
91: @torch.no_grad()
92: def collect_logits(model, loader, device) -> tuple[np.ndarray, np.ndarray]:
93:     """返回 (y_true, logits)，供 logit adjustment 在 val/test 上复用。"""
94:     model.eval()
95:     ys, zs = [], []
96:     for x, y in loader:
97:         x = x.to(device)
98:         zs.append(model(x).cpu().numpy())
99:         ys.append(y.numpy())
100:     return np.concatenate(ys), np.concatenate(zs)
101: 
102: 
103: def apply_logit_adjust(logits: np.ndarray, prior: np.ndarray, tau: float) -> np.ndarray:
104:     """post-hoc logit adjustment：z - tau * log(pi)。"""
105:     return logits - tau * np.log(np.clip(prior, 1e-12, None))
106: 
107: 
108: def tune_tau(logits: np.ndarray, y_true: np.ndarray, prior: np.ndarray,
109:              grid=(0.0, 0.25, 0.5, 0.75, 1.0)) -> tuple[float, dict]:
110:     """只在 val 上选 tau；返回 (tau, 对应 val 指标)。"""
111:     best_tau, best = 0.0, {"qwk": -2.0}
112:     for tau in grid:
113:         pred = apply_logit_adjust(logits, prior, tau).argmax(1)
114:         m = metrics_from_preds(y_true, pred)
115:         if m["qwk"] > best["qwk"] + 1e-12:
116:             best_tau, best = float(tau), m
117:     return best_tau, best
118: 
119: 
120: def class_counts(items) -> np.ndarray:
121:     return np.bincount([y for _, y in items], minlength=NUM_CLASSES).astype(np.float64)
122: 
123: 
124: def make_balanced_sampler(items) -> WeightedRandomSampler:
125:     """按逆频率采样，供 DRC 阶段2 用。"""
126:     cnt = class_counts(items)
127:     w = torch.tensor([1.0 / cnt[y] for _, y in items], dtype=torch.double)
128:     return WeightedRandomSampler(w, num_samples=len(w), replacement=True)
129: 
130: 
131: def freeze_backbone_reset_head(model: nn.Module, device: str) -> nn.Module:
132:     """DRC 阶段2：冻结全部参数，重初始化并只训练 fc。"""
133:     for p in model.parameters():
134:         p.requires_grad = False
135:     model.fc = nn.Linear(model.fc.in_features, NUM_CLASSES).to(device)
136:     for p in model.fc.parameters():
137:         p.requires_grad = True
138:     return model
139: 
140: 
141: @torch.no_grad()
142: def evaluate(model, loader, device) -> tuple[dict, np.ndarray, np.ndarray]:
143:     y_true, z = collect_logits(model, loader, device)
144:     y_pred = z.argmax(1)
145:     return metrics_from_preds(y_true, y_pred), y_true, y_pred
146: 
147: 
148: def train_one(size: int, root: Path, seed: int, epochs: int, bs: int,
149:               lr: float, weight_decay: float, device: str, weighted: bool,
150:               num_workers: int, quiet: bool, logit_adjust: bool = False,
151:               drc: bool = False, drc_stage2_epochs: int = 10,
152:               drc_stage2_lr: float = 1e-3) -> dict:
153:     random.seed(seed); np.random.seed(seed); torch.manual_seed(seed)
154: 
155:     tr = FundusFolder(root, "train", size, train=True)
156:     va = FundusFolder(root, "val", size, train=False)
157:     te = FundusFolder(root, "test", size, train=False)
158:     if not tr.items or not te.items:
159:         raise RuntimeError(f"数据为空：train={len(tr)} test={len(te)}")
160: 
161:     dl_tr = DataLoader(tr, batch_size=bs, shuffle=True, num_workers=num_workers,
162:                        pin_memory=True, drop_last=False)
163:     dl_va = DataLoader(va, batch_size=bs, shuffle=False, num_workers=num_workers)
164:     dl_te = DataLoader(te, batch_size=bs, shuffle=False, num_workers=num_workers)
165: 
166:     model = build_model(size).to(device)
167:     opt = torch.optim.AdamW(model.parameters(), lr=lr, weight_decay=weight_decay)
168:     sched = torch.optim.lr_scheduler.CosineAnnealingLR(opt, T_max=epochs)
169: 
170:     if weighted:
171:         cnt = class_counts(tr.items).astype(np.float32)
172:         w = torch.tensor((cnt.sum() / (cnt + 1e-6)) / NUM_CLASSES, device=device)
173:         crit = nn.CrossEntropyLoss(weight=w)
174:     else:
175:         crit = nn.CrossEntropyLoss()
176: 
177:     best = {"qwk": -2.0}
178:     best_state = None
179:     for ep in range(1, epochs + 1):
180:         model.train()
181:         tot = 0.0
182:         for x, y in dl_tr:
183:             x, y = x.to(device), y.to(device)
184:             opt.zero_grad(set_to_none=True)
185:             loss = crit(model(x), y)
186:             loss.backward(); opt.step()
187:             tot += loss.item() * len(y)
188:         sched.step()
189:         if ep % 5 == 0 or ep == epochs:
190:             m, _, _ = evaluate(model, dl_va, device) if len(va) else ({"qwk": 0}, None, None)
191:             if not quiet:
192:                 print(f"      seed{seed} ep{ep:3d}  train_loss {tot/len(tr):.4f}  val_qwk {m['qwk']:.4f}")
193:             if m["qwk"] > best["qwk"]:
194:                 best = m
195:                 best_state = {k: v.detach().cpu().clone() for k, v in model.state_dict().items()}
196: 
197:     if best_state is not None:
198:         model.load_state_dict(best_state)
199: 
200:     extra = {"mode": "drc" if drc else ("logit_adjust" if logit_adjust else "a0")}
201: 
202:     # ---- A2：DRC 阶段2——冻结骨干，重初始化分类头，类别平衡采样 ----
203:     if drc:
204:         freeze_backbone_reset_head(model, device)
205:         dl_bal = DataLoader(tr, batch_size=bs, sampler=make_balanced_sampler(tr.items),
206:                             num_workers=num_workers, pin_memory=True, drop_last=False)
207:         opt2 = torch.optim.AdamW(model.fc.parameters(), lr=drc_stage2_lr,
208:                                  weight_decay=weight_decay)
209:         crit2 = nn.CrossEntropyLoss()   # 采样已平衡，不再加权
210:         best2 = {"qwk": -2.0}
211:         best2_state = None
212:         for ep in range(1, drc_stage2_epochs + 1):
213:             model.train()
214:             # 冻结 BN 统计：backbone 保持 eval，只让 fc 更新
215:             for m_ in model.modules():
216:                 if isinstance(m_, nn.BatchNorm2d):
217:                     m_.eval()
218:             tot = 0.0
219:             for x, y in dl_bal:
220:                 x, y = x.to(device), y.to(device)
221:                 opt2.zero_grad(set_to_none=True)
222:                 loss = crit2(model(x), y)
223:                 loss.backward(); opt2.step()
224:                 tot += loss.item() * len(y)
225:             if len(va):
226:                 m, _, _ = evaluate(model, dl_va, device)
227:                 if not quiet:
228:                     print(f"      seed{seed} drc-ep{ep:3d}  loss {tot/len(tr):.4f}  val_qwk {m['qwk']:.4f}")
229:                 if m["qwk"] > best2["qwk"]:
230:                     best2 = m
231:                     best2_state = {k: v.detach().cpu().clone() for k, v in model.state_dict().items()}
232:         if best2_state is not None:
233:             model.load_state_dict(best2_state)
234:         extra["drc_stage2_epochs"] = drc_stage2_epochs
235:         extra["drc_stage2_lr"] = drc_stage2_lr
236:         extra["drc_val_qwk"] = best2["qwk"]
237: 
238:     # ---- A1：post-hoc logit adjustment（tau 只在 val 上选）／测试集只评一次 ----
239:     if logit_adjust:
240:         prior = class_counts(tr.items) / class_counts(tr.items).sum()
241:         if len(va):
242:             y_va, z_va = collect_logits(model, dl_va, device)
243:             tau, va_m = tune_tau(z_va, y_va, prior)
244:             extra["tau"] = tau
245:             extra["val_qwk_adjusted"] = va_m["qwk"]
246:         else:
247:             tau = 0.0
248:             extra["tau"] = tau
249:         y_true, z_te = collect_logits(model, dl_te, device)
250:         te_m = metrics_from_preds(y_true, apply_logit_adjust(z_te, prior, tau).argmax(1))
251:     else:
252:         te_m, y_true, y_pred = evaluate(model, dl_te, device)
253: 
254:     te_m["seed"] = seed
255:     te_m["val_qwk"] = best["qwk"] if not drc else extra.get("drc_val_qwk", best["qwk"])
256:     te_m["train_n"] = len(tr)
257:     te_m["test_n"] = len(te)
258:     te_m.update(extra)
259:     return te_m
260: 
261: 
262: def main() -> int:
263:     ap = argparse.ArgumentParser(description="HR 分级分类器（分辨率消融）")
264:     ap.add_argument("--data", required=True, help="正方形数据集根目录")
265:     ap.add_argument("--size", type=int, required=True, help="输入分辨率")
266:     ap.add_argument("--seeds", type=int, nargs="+", default=[42, 52, 62])
267:     ap.add_argument("--epochs", type=int, default=30)
268:     ap.add_argument("--batch-size", type=int, default=16)
269:     ap.add_argument("--lr", type=float, default=1e-4)
270:     ap.add_argument("--weight-decay", type=float, default=1e-4)
271:     ap.add_argument("--no-weighted", action="store_true", help="不加类别权重")
272:     ap.add_argument("--no-pretrained", action="store_true", help="不用 ImageNet 预训练")
273:     ap.add_argument("--logit-adjust", action="store_true",
274:                     help="A1：post-hoc logit adjustment，tau 只在 val 上选")
275:     ap.add_argument("--drc", action="store_true",
276:                     help="A2：DRC 两阶段（冻结骨干 + 类平衡重训分类头）")
277:     ap.add_argument("--drc-stage2-epochs", type=int, default=10,
278:                     help="DRC 阶段2 轮数（默认 10）")
279:     ap.add_argument("--drc-stage2-lr", type=float, default=1e-3,
280:                     help="DRC 阶段2 分类头学习率（默认 1e-3）")
281:     ap.add_argument("--num-workers", type=int, default=4)
282:     ap.add_argument("--out", default=None, help="结果 JSON 路径")
283:     ap.add_argument("--quiet", action="store_true")
284:     args = ap.parse_args()
285: 
286:     root = Path(args.data)
287:     if not root.is_dir():
288:         print(f"错误：数据目录不存在 {root}\n先跑 prep_square.py", file=sys.stderr)
289:         return 1
290: 
291:     device = "cuda" if torch.cuda.is_available() else "cpu"
292:     weighted = not args.no_weighted
293: 
294:     print(f"{'='*66}")
295:     print(f"  分辨率 {args.size}×{args.size}   设备 {device}   种子 {args.seeds}")
296:     print(f"  数据 {root}")
297:     print(f"{'='*66}")
298: 
299:     runs = []
300:     for s in args.seeds:
301:         t0 = time.time()
302:         try:
303:             r = train_one(args.size, root, s, args.epochs, args.batch_size, args.lr,
304:                           args.weight_decay, device, weighted, args.num_workers, args.quiet,
305:                           logit_adjust=args.logit_adjust, drc=args.drc,
306:                           drc_stage2_epochs=args.drc_stage2_epochs,
307:                           drc_stage2_lr=args.drc_stage2_lr)
308:         except RuntimeError as e:
309:             if "out of memory" in str(e).lower():
310:                 print(f"  seed{s} ❌ 显存不足，跳过")
311:                 torch.cuda.empty_cache()
312:                 continue
313:             raise
314:         runs.append(r)
315:         print(f"  seed{s:<3d} → acc {r['acc']:.4f}  bAcc {r['balanced_acc']:.4f}  "
316:               f"macroF1 {r['macro_f1']:.4f}  QWK {r['qwk']:.4f}  "
317:               f"severe召回 {r['severe_recall']:.4f}   ({time.time()-t0:.0f}s)")
318: 
319:     if not runs:
320:         print("  全部失败")
321:         return 2
322: 
323:     def agg(k):
324:         v = np.array([r[k] for r in runs], dtype=float)
325:         return float(v.mean()), float(v.std())
326: 
327:     summary = {k: agg(k) for k in ("acc", "balanced_acc", "macro_f1", "qwk", "severe_recall")}
328:     print(f"\n  ── 分辨率 {args.size} 汇总（{len(runs)} 个种子）──")
329:     for k, (m, s) in summary.items():
330:         print(f"     {k:16s} {m:.4f} ± {s:.4f}")
331: 
332:     if args.out:
333:         Path(args.out).parent.mkdir(parents=True, exist_ok=True)
334:         with open(args.out, "w", encoding="utf-8") as f:
335:             json.dump({"size": args.size, "seeds": args.seeds, "weighted": weighted,
336:                        "pretrained": not args.no_pretrained, "epochs": args.epochs,
337:                        "logit_adjust": args.logit_adjust, "drc": args.drc,
338:                        "drc_stage2_epochs": args.drc_stage2_epochs,
339:                        "summary": {k: {"mean": m, "std": s} for k, (m, s) in summary.items()},
340:                        "runs": runs}, f, ensure_ascii=False, indent=2)
341:         print(f"\n  已写 {args.out}")
342:     return 0
343: 
344: 
345: if __name__ == "__main__":
346:     raise SystemExit(main())
